import {readFileSync,writeFileSync} from 'node:fs';
import {createHash} from 'node:crypto';
import {runtimeModule} from '../web-player/lib/attiny13a.js';

import {parseHex} from '../web-player/lib/hex.js';

const abi=runtimeABI(new WebAssembly.Instance(runtimeModule).exports,{resumable:true});
function code(bytes){let at=8;const leb=()=>{let n=0,s=0,b;do{b=bytes[at++];n|=(b&127)<<s;s+=7;}while(b&128);return n;};
 while(at<bytes.length){const id=bytes[at++],n=leb();if(id===10)return bytes.slice(at,at+n);at+=n;}throw Error('No code');}
const report={cases:[]};
for(const id of ['rhythmic','lumiere-stereo','hq4-chego-tiny']){
 const entry=entries.find(e=>e.id===id),binary=parseHex(entry.flashHex);
 const raw=transpileAVR(binary,abi,{resumable:true}),folded=transpileAVR(binary,abi,{resumable:true,foldBlockConstants:true});
 for(const level of ['O3','Oz']){
  const a=optimizeAVRWasm(raw,{level}),b=optimizeAVRWasm(folded,{level});
  const row={id,level,tool:a.optimization.tool,executableSha256:a.optimization.executableSha256,
   before:[a.bytes.length,b.bytes.length],codeBytes:[code(a.bytes).length,code(b.bytes).length],
   codeSha256:[hash(code(a.bytes)),hash(code(b.bytes))],inputSha256:[hash(raw.bytes),hash(folded.bytes)]};
  row.identicalCode=row.codeSha256[0]===row.codeSha256[1];report.cases.push(row);console.log(JSON.stringify(row));
 }
}
writeFileSync(new URL('./block-constants-optimizer-comparison.json',import.meta.url),JSON.stringify(report,null,2)+'\n');

import {test} from 'node:test';
import assert from 'node:assert/strict';
import {readFileSync} from 'node:fs';
import {ATtiny13A} from '../lib/attiny13a.js';
import {avr,IO,flash,sequential} from './fixtures/attiny13a-system-firmware.mjs';

const registers=['ADCSRB','ADMUX','DIDR0','PCMSK','DDRB','PORTB','MCUCR','GIMSK',
  'EEDR','EEARL','PRR','TCCR0A','TCCR0B','OCR0A','OCR0B','TIMSK0'];
const loop=flash({instructions:[avr.inc(16),avr.rjmp(1,0)]});
const hz=20000000,osccalHz=Array(128).fill(0);osccalHz[0]=hz;
const options=clock=>({oscillatorHz:hz,fuseLow:clock==='clki'?0x70:0x72,
  ...(clock==='physical'?{resumable:true,osccalHz}:{}),
  ...(clock==='clki'?{resumable:true,clkiProfile:{phasePolicy:'digital-noninverting-zero-seed-v1',initialLevel:1}}:{})});
const make=(clock,enabled,binary=loop)=>new ATtiny13A(binary,null,{...options(clock),optimizeIOPublication:enabled});
const snapshot=c=>({state:c.state(),diagnostics:Array.from(c._state),data:c.data.slice(),
  flash:c.rom.slice(),eeprom:c.eeprom.slice(),irq:Array.from(c.irqFrames),pins:c._ps?c.drainPinsPs():c.drainPins()});
const poison=c=>{for(const name of registers)c.data[32+IO[name]]^=255;};
const assertReleased=c=>{assert.equal(c._exports.avr_device_io_scope_begin(),1);c._exports.avr_device_io_scope_end();};

for(const clock of ['compatible','physical','clki']){
  test(`I/O scope preserves zero-work and restores raw shadows on execution: ${clock}`,()=>{
    const cores=[make(clock,false),make(clock,true)];
    for(const c of cores){
      c.writeIO(IO.DDRB,2);c.writeIO(IO.PORTB,1);c.writeIO(IO.OCR0A,0x71);
      poison(c);const before=c.data.slice();
      assert.equal(clock==='clki'?c._execute('run',0):c.run(0),0);assert.deepEqual(c.data,before);assertReleased(c);
      if(clock==='clki')c.injectClkiPeriodicPs({startPs:0n,initialLevel:1,
        highPs:25000n,lowPs:25000n,transitionCount:200n,knownUntilPs:5000000n});
      if(clock==='clki')c.runUntilPs(200000n);else c.run(10);
      assert.equal(c.data[32+IO.DDRB],2);assert.equal(c.data[32+IO.PORTB],1);
      assert.equal(c.data[32+IO.OCR0A],0x71);assertReleased(c);
    }
    assert.deepEqual(snapshot(cores[1]),snapshot(cores[0]));
  });
  test(`direct native/generated entry retains full clock publication: ${clock}`,()=>{
    const c=make(clock,true);c.writeIO(IO.PORTB,1);
    if(clock==='clki')c.injectClkiPeriodicPs({startPs:0n,initialLevel:1,
      highPs:25000n,lowPs:25000n,transitionCount:200n,knownUntilPs:5000000n});
    for(let n=0;n<2;n++){
      poison(c);
      if(clock==='clki'){
        c._exports.avr_device_phase_deadline(200000*(n+1));c._translated.runUntil(200000*(n+1));
      }else c._translated.run(10);
      assert.equal(c.data[32+IO.PORTB],1);assertReleased(c);
    }
  });
}

test('unknown CLKI WAIT and a zero physical deadline preserve unserviced shadows',()=>{
  for(const clock of ['physical','clki']){
    const cores=[make(clock,false),make(clock,true)];
    for(const c of cores){poison(c);c.runUntilPs(clock==='clki'?100000n:0n);assertReleased(c);}
    assert.deepEqual(snapshot(cores[1]),snapshot(cores[0]));
    assert.equal(cores[1].data[32+IO.PORTB],255);
  }
});

test('configuration cross-effects remain visible within a batch and at partial edges',()=>{
  const binary=sequential([avr.ldi(16,2),avr.out(IO.DDRB,16),avr.out(IO.PORTB,16),
    avr.out(IO.PINB,16),avr.in(17,IO.PORTB),avr.ldi(16,43),avr.out(IO.EEARL,16),
    avr.ldi(16,1),avr.out(IO.EECR,16),avr.in(18,IO.EEDR),avr.ldi(16,0x23),
    avr.out(IO.ADMUX,16),avr.in(20,IO.ADCL),avr.ldi(16,3),avr.out(IO.ADMUX,16),
    avr.ldi(16,0x71),avr.out(IO.OCR0A,16),avr.in(19,IO.OCR0A)]);
  for(const clock of ['compatible','physical','clki']){
    const cores=[make(clock,false,binary),make(clock,true,binary)];
    for(const c of cores){
      c.eeprom[43]=0x5a;c.writeIO(IO.ADMUX,3);c.writeIO(IO.ADCSRA,0xc7);
      if(clock==='clki')c.injectClkiPeriodicPs({startPs:0n,initialLevel:1,
        highPs:25000n,lowPs:25000n,transitionCount:4000n,knownUntilPs:100000000n});
    }
    const stops=clock==='compatible'?[1,9,100]:[1n,49999n,50000n,50001n,1000000n,10000000n];
    for(const stop of stops){
      for(const c of cores){poison(c);if(clock==='compatible')c.run(stop);else c.runUntilPs(stop);assertReleased(c);}
      assert.deepEqual(snapshot(cores[1]),snapshot(cores[0]),clock+' boundary '+stop);
    }
    assert.equal(cores[1].r[17],0);assert.equal(cores[1].r[18],0x5a);assert.equal(cores[1].r[19],0x71);
  }
});

test('scope cleanup survives a real WASM trap and a pre-execution host exception',()=>{
  const trap=new WebAssembly.Instance(new WebAssembly.Module(Uint8Array.from([
    0,97,115,109,1,0,0,0,1,4,1,96,0,0,3,2,1,0,7,8,1,4,116,114,97,112,0,0,10,5,1,3,0,0,11]))).exports.trap;
  for(const before of [false,true]){
    const c=make('physical',true),original=c._bridges.run.translated,call=c._call;
    if(before)c._call=()=>{throw Error('before execution');};else c._bridges.run.translated=trap;
    poison(c);const data=c.data.slice();assert.throws(()=>c.run(1),before?/before execution/:WebAssembly.RuntimeError);
    assert.deepEqual(c.data,data);assertReleased(c);
    c._call=call;c._bridges.run.translated=original;c.run(1);assert.equal(c.data[32+IO.PORTB],0);
  }
});

test('nested native scope admission fails without publishing or releasing the owner',()=>{
  const c=make('compatible',true),e=c._exports;poison(c);const before=c.data.slice();
  assert.equal(e.avr_device_io_scope_begin(),1);assert.equal(e.avr_device_io_scope_begin(),0);
  assert.throws(()=>c.run(1),/scope already active/);assert.deepEqual(c.data,before);
  assert.equal(e.avr_device_io_scope_begin(),0);e.avr_device_io_scope_end();
  assert.deepEqual(c.data,before);c.run(1);assertReleased(c);
});

for(const power of [false,true])test(`reset cancels a partial store and republishes configuration: power=${power}`,()=>{
  // Explicit synthetic board thresholds, shared with the power integration tests.
  const powerResetProfile={porAssertMicrovolts:1300000,porReleaseMicrovolts:1400000,
    bod00Microvolts:4300000,bod01Microvolts:2700000,bod10Microvolts:1800000,
    bodHysteresisMicrovolts:50000,bodFilterNs:2000,bodRecoveryNs:60000,flashProgrammingMinMicrovolts:1800000};
  const binary=sequential([avr.ldi(16,0x5a),avr.sts(0x60,16)]);
  const cores=[false,true].map(enabled=>new ATtiny13A(binary,null,{...options('physical'),
    optimizeIOPublication:enabled,...power?{powerResetProfile}:{}}));
  for(const c of cores){
    c.step();c.writeIO(IO.PORTB,3);poison(c);
    const at=c.state().timePs+37n;
    if(power)c.setVccPs({at,microvolts:0});else c.injectPinsPs({at,mask:32,levels:0});
    c.runUntilPs(200000n);assertReleased(c);
    assert.equal(c.data[0x60],0);assert.equal(c.state().resetCount,1);
    assert.equal(c.data[32+IO.PORTB],0);assert.equal(c.state().resetHeld,true);
  }
  assert.deepEqual(snapshot(cores[1]),snapshot(cores[0]));
  for(const c of cores){
    if(power)c.setVccPs({at:250000n,microvolts:5000000});else c.injectPinsPs({at:250000n,mask:32,levels:32});
    poison(c);c.runUntilPs(300000n);assertReleased(c);
  }
  assert.deepEqual(snapshot(cores[1]),snapshot(cores[0]));
});

test('peripheral faults inside execution and an existing CPU error release the scope',()=>{
  const binary=sequential([avr.ldi(16,7),avr.out(IO.ADCSRB,16)]);
  const cores=[make('physical',false,binary),make('physical',true,binary)];
  for(const c of cores){assert.throws(()=>c.run(2),/peripheral fault 4, detail 7/);assertReleased(c);}
  assert.deepEqual(snapshot(cores[1]),snapshot(cores[0]));
  for(const enabled of [false,true]){
    const c=make('physical',enabled);c.error=1;poison(c);const data=c.data.slice();
    assert.throws(()=>c.run(1));assert.deepEqual(c.data,data);assertReleased(c);
  }
});

// Rewrite only export indices/names; leave every native function body intact.
const runtime=readFileSync(new URL('../wasm/attiny13a.wasm',import.meta.url));
const leb=n=>{const a=[];do{const b=n&127;n>>>=7;a.push(b|(n?128:0));}while(n);return a;};
function alteredBinary(changes){
  let at=8;const read=()=>{let n=0,s=0,b;do{b=runtime[at++];n+=(b&127)*2**s;s+=7;}while(b&128);return n;};
  while(at<runtime.length){
    const start=at,id=runtime[at++],length=read(),end=at+length;
    if(id!==7){at=end;continue;}
    const rows=Array.from({length:read()},()=>{const size=read(),name=runtime.subarray(at,at+=size).toString();return {name,kind:runtime[at++],index:read()};});
    assert.equal(at,end);
    const entries=rows.filter(r=>changes[r.name]!==null).map(r=>({ ...r,index:changes[r.name]?rows.find(x=>x.name===changes[r.name]).index:r.index}));
    const payload=[...leb(entries.length),...entries.flatMap(r=>{const b=Buffer.from(r.name);return [...leb(b.length),...b,r.kind,...leb(r.index)];})];
    return Buffer.concat([runtime.subarray(0,start),Buffer.from([7,...leb(payload.length),...payload]),runtime.subarray(end)]);
  }
  throw Error('missing export section');
}
const alteredExports=changes=>new WebAssembly.Module(alteredBinary(changes));
test('optional scope validates exact WASM signatures and historical absence',()=>{
  const begin='avr_device_io_scope_begin',end='avr_device_io_scope_end';
  for(const changes of [{[begin]:end},{[end]:begin},{[end]:null}])
    assert.throws(()=>new ATtiny13A(loop,null,{runtimeModule:alteredExports(changes)}),/I\/O publication scope ABI/);
  const c=new ATtiny13A(loop,null,{runtimeModule:alteredExports({[begin]:null,[end]:null})});
  poison(c);c.run(2);assert.equal(c.data[32+IO.PORTB],0);assert.equal(c._ioScope,null);
  assert.throws(()=>new ATtiny13A(loop,null,{optimizeIOPublication:1}),/requires a boolean/);
});
test('package validation rejects incomplete or mistyped optional scope exports',async()=>{
  const {createPackage}=await import('../lib/attiny13a-package.js');
  const c=make('compatible',true),begin='avr_device_io_scope_begin',end='avr_device_io_scope_end';
  for(const changes of [{[begin]:end},{[end]:begin},{[end]:null}])
    await assert.rejects(createPackage({flash:loop,translated:c.transpiledBytes,runtime:alteredBinary(changes)}),/runtime export mismatch: avr_device_io_scope_/);
});
